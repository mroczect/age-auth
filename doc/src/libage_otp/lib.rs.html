1//! # libage_otp
2//!
3//! Pure Rust implementation of TOTP (RFC 6238) and HOTP (RFC 4226).
4//!
5//! This crate provides the [`AgeOtp`] struct, which implements the
6//! [`OtpGenerator`] trait from [`libage_auth_handler`].
7//! It is entirely independent of the age encryption layer – it only
8//! deals with raw secrets and counters/timestamps.
9//!
10//! # Quick start
11//!
12//! ```rust
13//! use libage_otp::AgeOtp;
14//! use libage_auth_handler::traits::OtpGenerator;
15//! use libage_auth_handler::types::{Secret, Digits, Counter, TimeStep};
16//! use libage_auth_handler::Algo;
17//!
18//! # fn main() -> Result<(), Box<dyn std::error::Error>> {
19//! let secret = Secret::new(b"12345678901234567890".to_vec());
20//!
21//! // HOTP
22//! let token = AgeOtp::hotp(&secret, Counter::new(0), Digits::new(6)?, Algo::Sha1)?;
23//! println!("HOTP: {}", token.format(6));
24//!
25//! // TOTP (current time)
26//! let token = AgeOtp::totp(&secret, TimeStep::default(), Digits::default(), Algo::Sha256)?;
27//! println!("TOTP: {}", token.format(6));
28//! # Ok(())
29//! # }
30//! ```

39/// OTP generator implementing [`OtpGenerator`].
40///
41/// `AgeOtp` is a zero‑sized struct that delegates to the functions in
42/// the [`algorithms`] module. All methods are associated functions
43/// (they do not take `&self`), so they are called as `AgeOtp::totp(...)`.
44///
45/// # Security
46///
47/// - The secret is accessed only via `&[u8]` and never cloned.
48/// - HMAC results are dropped immediately after truncation.
49/// - No `unwrap`/`expect` is used on user input.
50/// - The system clock is read once per call; clock errors are returned,
51///   not panicked.
52///
53/// # Examples
54///
55/// ```rust
56/// use libage_otp::AgeOtp;
57/// use libage_auth_handler::traits::OtpGenerator;
58/// use libage_auth_handler::types::{Secret, Counter, Digits};
59/// use libage_auth_handler::Algo;
60///
61/// let secret = Secret::new(b"12345678901234567890".to_vec());
62/// let token = AgeOtp::hotp(&secret, Counter::new(0), Digits::new(6).unwrap(), Algo::Sha1)
63///     .expect("HOTP generation failed");
64/// assert_eq!(token.format(6).len(), 6);
65/// ```
66pub struct AgeOtp;
67
68impl OtpGenerator for AgeOtp {
69    /// Generates a time‑based one‑time password (TOTP).
70    ///
71    /// Uses the current system time as the counter. For a deterministic
72    /// version, use [`algorithms::compute_totp_at`] directly.
73    ///
74    /// # Parameters
75    /// - `secret`: The shared secret.
76    /// - `time_step`: Time step in seconds (must be > 0).
77    /// - `digits`: Number of digits for the token (4‑10).
78    /// - `algo`: HMAC algorithm ([`Algo::Sha1`], [`Algo::Sha256`], [`Algo::Sha512`]).
79    ///
80    /// # Errors
81    /// Returns [`libage_auth_handler::errors::AuthError::Otp`] if the system clock is before the Unix
82    /// epoch or the computed token exceeds the allowed range.
83    ///
84    /// # Example
85    /// ```rust
86    /// # use libage_otp::AgeOtp;
87    /// # use libage_auth_handler::traits::OtpGenerator;
88    /// # use libage_auth_handler::types::{Secret, TimeStep, Digits};
89    /// # use libage_auth_handler::Algo;
90    /// let secret = Secret::new(b"12345678901234567890".to_vec());
91    /// let token = AgeOtp::totp(
92    ///     &secret,
93    ///     TimeStep::new(30).unwrap(),
94    ///     Digits::new(6).unwrap(),
95    ///     Algo::Sha256,
96    /// ).expect("TOTP failed");
97    /// assert_eq!(token.format(6).len(), 6);
98    /// ```
99    fn totp(secret: &Secret, time_step: TimeStep, digits: Digits, algo: Algo) -> Result<Token> {
100        let now = algorithms::current_unix_time()?;
101        algorithms::compute_totp_at(secret, now, time_step, digits, algo)
102    }
103
104    /// Generates a counter‑based one‑time password (HOTP).
105    ///
106    /// See [`algorithms::compute_hotp_at`] for the underlying algorithm.
107    ///
108    /// # Parameters
109    /// - `secret`: The shared secret.
110    /// - `counter`: Counter value (8‑byte big‑endian integer internally).
111    /// - `digits`: Number of digits for the token (4‑10).
112    /// - `algo`: HMAC algorithm.
113    ///
114    /// # Errors
115    /// Returns [`libage_auth_handler::errors::AuthError::Otp`] if the secret is invalid for the chosen
116    /// algorithm or the computed token exceeds the allowed range.
117    ///
118    /// # Example
119    /// ```rust
120    /// # use libage_otp::AgeOtp;
121    /// # use libage_auth_handler::traits::OtpGenerator;
122    /// # use libage_auth_handler::types::{Secret, Counter, Digits};
123    /// # use libage_auth_handler::Algo;
124    /// let secret = Secret::new(b"12345678901234567890".to_vec());
125    /// let token = AgeOtp::hotp(
126    ///     &secret,
127    ///     Counter::new(42),
128    ///     Digits::new(8).unwrap(),
129    ///     Algo::Sha256,
130    /// ).expect("HOTP failed");
131    /// assert_eq!(token.format(8).len(), 8);
132    /// ```
133    fn hotp(secret: &Secret, counter: Counter, digits: Digits, algo: Algo) -> Result<Token> {
134        algorithms::compute_hotp_at(secret, counter.value(), digits, algo)
135    }
136
137    /// Convenience method: decode a Base32‑encoded secret and return the
138    /// current TOTP code using default parameters.
139    ///
140    /// Defaults: time step = **30s**, digits = **6**, algorithm = **SHA‑256**.
141    ///
142    /// # Parameters
143    /// - `secret_base32`: A valid Base32‑encoded secret string (RFC 4648,
144    ///   no padding).
145    ///
146    /// # Returns
147    /// A formatted OTP string (e.g., `"123456"`).
148    ///
149    /// # Errors
150    /// Returns `Err` if:
151    /// - The Base32 string is invalid or cannot be decoded.
152    /// - The system clock is before the Unix epoch.
153    /// - The generated token exceeds 6 digits (should never occur).
154    ///
155    /// # Example
156    /// ```rust
157    /// # use libage_otp::AgeOtp;
158    /// # use libage_auth_handler::traits::OtpGenerator;
159    /// # use libage_auth_handler::types::Base32String;
160    /// let base32 = Base32String::new("JBSWY3DPEHPK3PXP").unwrap();
161    /// let code = AgeOtp::totp_now_from_base32(&base32).unwrap();
162    /// assert_eq!(code.len(), 6);
163    /// ```
164    fn totp_now_from_base32(secret_base32: &Base32String) -> Result<String> {
165        let secret = secret_base32.to_secret()?;
166        let time_step = TimeStep::default();
167        let digits = Digits::default();
168        let token = Self::totp(&secret, time_step, digits, Algo::DEFAULT)?;
169        Ok(token.format(digits.value()))
170    }
171}
