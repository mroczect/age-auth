1//! Unified error type for the `age-auth` workspace.
2//!
3//! All fallible functions in `libage_auth_handler`, `libage_crypto`,
4//! `libage_otp`, and `libage_authenticator` return
5//! [`Result<T>`](crate::errors::Result) which is an alias for
6//! `std::result::Result<T, AuthError>`.
7//!
8//! `AuthError` is designed to be descriptive, easy to match, and to
9//! automatically convert from `std::io::Error` and other error types.
10
11use thiserror::Error;
12
13/// The single error type for all `age-auth` crates.
14///
15/// # Variants
16/// - `Crypto(String)` – cryptographic operation failed (wrong key, MAC
17///   mismatch, etc.).
18/// - `Otp(String)` – OTP generation failed (e.g., invalid secret or
19///   unsupported algorithm).
20/// - `InvalidInput(String)` – user input did not pass validation
21///   (malformed recipient, identity, base32 string, …).
22/// - `Io(std::io::Error)` – file or I/O error. Transparently wraps
23///   `std::io::Error`.
24/// - `Other(Box<dyn Error + Send + Sync>)` – catch‑all for rare or
25///   unexpected errors.
26///
27/// # `Display` and `Error`
28/// `AuthError` implements `std::error::Error` and provides human‑readable
29/// messages via `Display` (powered by `thiserror`).
30///
31/// # Conversions
32/// * `From<io::Error>` → `AuthError::Io`
33/// * `From<Box<dyn Error + Send + Sync>>` → `AuthError::Other`
34///
35/// This allows the `?` operator to be used directly with I/O errors.
36///
37/// # Examples
38/// ```rust
39/// use libage_auth_handler::errors::AuthError;
40///
41/// let err = AuthError::InvalidInput("bad recipient".into());
42/// println!("{}", err); // prints "Invalid input: bad recipient"
43///
44/// let io_err = std::io::Error::new(std::io::ErrorKind::NotFound, "file missing");
45/// let auth_err: AuthError = io_err.into();
46/// assert!(matches!(auth_err, AuthError::Io(_)));
47/// ```
48
49#[derive(Error, Debug)]
50pub enum AuthError {
51    /// Cryptography‑related failure (e.g., decryption failed, key rejected).
52    #[error("Crypto error: {0}")]
53    Crypto(String),
54
55    /// OTP generation failure (e.g., algorithm not supported, invalid secret).
56    #[error("OTP error: {0}")]
57    Otp(String),
58
59    /// Input validation failure (e.g., malformed recipient, identity, or base32
60    /// string).
61    #[error("Invalid input: {0}")]
62    InvalidInput(String),
63
64    /// I/O error. Wraps `std::io::Error` transparently.
65    #[error("I/O error: {0}")]
66    Io(#[from] std::io::Error),
67
68    /// Any other error that is not covered by the specific variants.
69    #[error("Unknown error: {0}")]
70    Other(#[from] Box<dyn std::error::Error + Send + Sync>),
71}
72
73/// Convenience type alias for `Result<T, AuthError>`.
74///
75/// # Examples
76/// ```rust
77/// use libage_auth_handler::errors::{AuthError, Result};
78///
79/// fn might_fail(ok: bool) -> Result<()> {
80///     if ok {
81///         Ok(())
82///     } else {
83///         Err(AuthError::InvalidInput("something wrong".into()))
84///     }
85/// }
86/// ```
87pub type Result<T> = std::result::Result<T, AuthError>;
