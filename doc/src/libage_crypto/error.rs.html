1//! Error mapping from `librage` to the workspace's `AuthError`.
2//!
3//! This module contains a single helper function that translates
4//! [`librage::ErrorBody`] (a structured error with a machine-readable
5//! `code` and a human-readable `message`) into the unified
6//! [`AuthError`](libage_auth_handler::errors::AuthError) enum used
7//! throughout the `age-auth` workspace.
8//!
9//! The mapping follows these rules:
10//!
11//! | `librage` error code                                     | `AuthError` variant         |
12//! | -------------------------------------------------------- | --------------------------- |
13//! | `INVALID_PUBLIC_KEY`, `INVALID_SSH_KEY`, `INVALID_INPUT` | `InvalidInput(String)`      |
14//! | `INVALID_SECRET_KEY`                                     | `Crypto(String)`            |
15//! | `IO_ERROR`                                               | `Io(std::io::Error)`        |
16//! | Any other code                                           | `Crypto(String)`            |
17//!
18//! This ensures that key-related input errors (invalid public/SSH keys)
19//! are reported as user‑facing `InvalidInput`, while a bad secret key
20//! is treated as a cryptographic error (`Crypto`), because the secret
21//! key is not typically entered by the end user but rather represents a
22//! configuration or key‑management problem.
23
24use libage_auth_handler::errors::AuthError;
25
26/// Converts a [`librage::ErrorBody`] into an [`AuthError`].
27///
28/// The function examines the `code` field of the error body and produces
29/// an appropriate `AuthError` variant. This is the single point of
30/// translation between `librage`'s error model and the workspace's
31/// unified error type.
32///
33/// # Arguments
34///
35/// * `body` - A reference to the error body returned by a failed
36///   `librage` operation.
37///
38/// # Examples
39///
40/// ```ignore
41/// # use libage_auth_handler::errors::AuthError;
42/// # use libage_crypto::error::from_librage_error_body;
43/// let body = librage::ErrorBody::new("INVALID_PUBLIC_KEY", "bad key");
44/// let err = from_librage_error_body(&body);
45/// assert!(matches!(err, AuthError::InvalidInput(_)));
46///
47/// let body = librage::ErrorBody::new("INVALID_SECRET_KEY", "wrong secret");
48/// let err = from_librage_error_body(&body);
49/// assert!(matches!(err, AuthError::Crypto(_)));
50/// ```
51pub(crate) fn from_librage_error_body(body: &librage::ErrorBody) -> AuthError {
52    match body.code.as_str() {
53        "INVALID_PUBLIC_KEY" | "INVALID_SSH_KEY" | "INVALID_INPUT" => {
54            AuthError::InvalidInput(body.message.clone())
55        }
56        "INVALID_SECRET_KEY" => AuthError::Crypto(format!("Invalid secret key: {}", body.message)),
57        "IO_ERROR" => AuthError::Io(std::io::Error::other(body.message.clone())),
58        _ => AuthError::Crypto(format!("{}: {}", body.code, body.message)),
59    }
60}
