21const RECIPIENT_PREFIX: bech32::Hrp = bech32::Hrp::parse_unchecked("age1tagpq");
22
23const MLKEM768P256TAG_RECIPIENT_TAG: &str = "mlkem768p256tag";
24const MLKEM768P256TAG_SALT: &str = "age-encryption.org/mlkem768p256tag";
25
26type Kem = kem::MlKem768P256;
27
28pub(crate) fn expand_pq_key(
29    seed: &[u8; 64],
30) -> (
31    <MlKem768 as KemCore>::DecapsulationKey,
32    <MlKem768 as KemCore>::EncapsulationKey,
33) {
34    let mut d = [0; 32];
35    let mut z = [0; 32];
36    d.copy_from_slice(&seed[..32]);
37    z.copy_from_slice(&seed[32..]);
38
39    let (dk_pq, ek_pq) = MlKem768::generate_deterministic(&d.into(), &z.into());
40
41    d.zeroize();
42    z.zeroize();
43
44    (dk_pq, ek_pq)
45}

47/// The hybrid post-quantum tagged age recipient type, designed for hardware keys where
48/// decryption potentially requires user presence.
49///
50/// With knowledge of the recipient, it is possible to check if a stanza was addressed to
51/// a specific recipient before attempting decryption. This offers less privacy than the
52/// untagged recipient types.
53#[derive(Clone, PartialEq)]
54pub struct Recipient(<Kem as hpke::Kem>::PublicKey);
55
56impl std::str::FromStr for Recipient {
57    type Err = &'static str;
58
59    /// Parses a recipient key from a string.
60    fn from_str(s: &str) -> Result<Self, Self::Err> {
61        bech32_decode(
62            s,
63            |_| "invalid Bech32 encoding",
64            |hrp| {
65                (hrp == RECIPIENT_PREFIX)
66                    .then_some(())
67                    .ok_or("incorrect HRP")
68            },
69            |_, bytes| {
70                <Kem as hpke::Kem>::PublicKey::from_bytes(&bytes.collect::<Vec<_>>())
71                    .map_err(|_| "invalid recipient")
72                    .map(Self)
73            },
74        )
75    }
76}
77
78impl fmt::Display for Recipient {
79    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
80        bech32_encode_to_fmt(f, RECIPIENT_PREFIX, &self.0.to_bytes())
81    }
82}

90impl crate::Recipient for Recipient {
91    fn wrap_file_key(
92        &self,
93        file_key: &FileKey,
94    ) -> Result<(Vec<Stanza>, HashSet<String>), EncryptError> {
95        let (enc, ct) = hpke_seal::<Kem, _>(
96            &self.0,
97            MLKEM768P256TAG_SALT.as_bytes(),
98            file_key.expose_secret(),
99            &mut OsRng,
100        );
101
102        let ikm = enc
103            .to_bytes()
104            .into_iter()
105            .chain(super::static_tag(
106                self.0.ek_t.to_encoded_point(false).as_bytes(),
107            ))
108            .collect::<Vec<u8>>();
109        let tag = super::stanza_tag(&ikm, MLKEM768P256TAG_SALT);
110
111        let encoded_tag = BASE64_STANDARD_NO_PAD.encode(tag);
112        let encoded_enc = BASE64_STANDARD_NO_PAD.encode(enc.to_bytes());
113
114        Ok((
115            vec![Stanza {
116                tag: MLKEM768P256TAG_RECIPIENT_TAG.to_owned(),
117                args: vec![encoded_tag, encoded_enc],
118                body: ct,
119            }],
120            super::label_pq_only(),
121        ))
122    }
123}
