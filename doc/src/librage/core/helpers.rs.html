1use crate::handler::error::LibrageError;
2use age::{Decryptor, Encryptor};

5pub(crate) fn encrypt_to_writer<W: Write>(
6    plaintext: &[u8],
7    recipients: &[Box<dyn age::Recipient>],
8    mut writer: W,
9) -> Result<W, LibrageError> {
10    let encryptor = Encryptor::with_recipients(recipients.iter().map(|r| r.as_ref()))
11        .map_err(LibrageError::from)?;
12    let mut encrypt_writer = encryptor
13        .wrap_output(&mut writer)
14        .map_err(LibrageError::from)?;
15    encrypt_writer.write_all(plaintext)?;
16    encrypt_writer.finish().map_err(LibrageError::from)?;
17    Ok(writer)
18}

20pub(crate) fn decrypt_to_vec<R: Read>(
21    reader: R,
22    identities: &[Box<dyn age::Identity>],
23) -> Result<Vec<u8>, LibrageError> {
24    use std::io::BufReader;
25    let decryptor = Decryptor::new_buffered(BufReader::new(reader)).map_err(LibrageError::from)?;
26    let mut plaintext = Vec::new();
27    let mut plain_reader = decryptor
28        .decrypt(identities.iter().map(|i| i.as_ref()))
29        .map_err(LibrageError::from)?;
30    plain_reader.read_to_end(&mut plaintext)?;
31    Ok(plaintext)
32}

60pub(crate) fn encrypt_stream_to_armored_writer<R: Read, W: Write>(
61    mut reader: R,
62    writer: W,
63    recipients: &[Box<dyn age::Recipient>],
64) -> Result<(), LibrageError> {
65    let encryptor = Encryptor::with_recipients(recipients.iter().map(|r| r.as_ref()))
66        .map_err(LibrageError::from)?;
67    let armored_writer =
68        age::armor::ArmoredWriter::wrap_output(writer, age::armor::Format::AsciiArmor)
69            .map_err(|e| LibrageError::EncryptionFailed(e.to_string()))?;
70    let mut encrypt_writer = encryptor
71        .wrap_output(armored_writer)
72        .map_err(LibrageError::from)?;
73    std::io::copy(&mut reader, &mut encrypt_writer)?;
74    let armored_writer = encrypt_writer.finish().map_err(LibrageError::from)?;
75    armored_writer.finish().map_err(LibrageError::from)?;
76    Ok(())
