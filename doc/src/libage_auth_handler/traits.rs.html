1//! Core traits for the `age-auth` workspace.
2//!
3//! This module defines the three main abstractions used throughout the
4//! workspace:
5//!
6//! - [`CryptoBackend`] – encryption/decryption of secrets.
7//! - [`OtpGenerator`] – TOTP/HOTP generation.
8//! - [`Authenticator`] – combines the above two to provide a high‑level
9//!   authenticator API.
10//!
11//! Implementations of these traits are provided by the other crates in the
12//! workspace (`libage_crypto`, `libage_otp`, `libage_authenticator`).
13
14use crate::enums::Algo;
15use crate::errors::Result;
16use crate::types::{
17    Base32String, Counter, Digits, EncryptedPayload, Identity, Recipient, Secret, TimeStep, Token,
18};
19use std::io::Read;

21/// Cryptographic backend that can encrypt and decrypt secrets using
22/// [age](https://age-encryption.org) identities.
23///
24/// # Implementors
25/// - `libage_crypto::AgeCrypto`
26/// - `libage_authenticator::AgeAuthenticator`
27///
28/// # Safety
29/// Implementations must guarantee that:
30/// - The plaintext secret is never exposed after the call.
31/// - The ciphertext can only be decrypted by the intended identity.
32pub trait CryptoBackend {
33    /// Encrypts `plaintext` for the given `recipient` (age public key).
34    ///
35    /// Returns the ciphertext wrapped in an [`EncryptedPayload`].
36    fn encrypt(&self, recipient: &Recipient, plaintext: &Secret) -> Result<EncryptedPayload>;
37
38    /// Decrypts `ciphertext` using the provided `identity` (age secret key).
39    ///
40    /// Returns the original [`Secret`].
41    fn decrypt(&self, identity: &Identity, ciphertext: &EncryptedPayload) -> Result<Secret>;
42}
43
44/// Generator for Time‑based (TOTP) and Counter‑based (HOTP) one‑time passwords.
45///
46/// All methods are **associated functions** (they do not take `&self`), so they
47/// are called directly on the implementor type, e.g. `AgeOtp::totp(...)`.
48///
49/// # Implementors
50/// - `libage_otp::AgeOtp`
51/// - `libage_authenticator::AgeAuthenticator`
52///
53/// # Algorithm
54/// - TOTP follows RFC 6238.
55/// - HOTP follows RFC 4226.
56/// - Dynamic truncation is applied according to RFC 4226 Section 5.3.
57pub trait OtpGenerator {
58    /// Generates a time‑based one‑time password.
59    ///
60    /// Uses the current system time. If the system clock is before the Unix
61    /// epoch, an error is returned.
62    fn totp(secret: &Secret, time_step: TimeStep, digits: Digits, algo: Algo) -> Result<Token>;
63
64    /// Generates a counter‑based one‑time password.
65    fn hotp(secret: &Secret, counter: Counter, digits: Digits, algo: Algo) -> Result<Token>;
66
67    /// Convenience method: decodes a Base32‑encoded secret, then calls `totp`
68    /// with default time step (30s), default digits (6), and the default
69    /// algorithm (`Sha256`). Returns the formatted token string.
70    fn totp_now_from_base32(secret_base32: &Base32String) -> Result<String>;
71}

73/// High‑level authenticator that combines encryption and OTP generation.
74///
75/// This trait is **automatically implemented** for any type that implements
76/// both `CryptoBackend` and `OtpGenerator`, providing a complete offline
77/// authenticator workflow.
78///
79/// # Provided methods
80/// - `provision` – encrypt a secret.
81/// - `load_encrypted_secret` – decrypt an encrypted secret.
82/// - `generate_totp_from_encrypted` – decrypt and immediately generate a
83///   TOTP code.
84///
85/// # Implementors
86/// - `libage_authenticator::AgeAuthenticator`
87pub trait Authenticator: CryptoBackend + OtpGenerator {
88    /// Encrypts `secret` with `public_key`.
89    ///
90    /// This is a convenience wrapper around `CryptoBackend::encrypt`.
91    fn provision(&self, public_key: &Recipient, secret: &Secret) -> Result<EncryptedPayload> {
92        self.encrypt(public_key, secret)
93    }
94
95    /// Reads an identity from `identity_reader`, decrypts `encrypted`, and
96    /// returns the resulting secret.
97    ///
98    /// The identity is read into a zeroizing buffer to prevent it from
99    /// lingering in memory.
100    fn load_encrypted_secret(
101        &self,
102        identity_reader: &mut dyn Read,
103        encrypted: &EncryptedPayload,
104    ) -> Result<Secret>;
105
106    /// Full pipeline: decrypts the secret, interprets it as a UTF‑8 Base32
107    /// string, decodes it, and returns the current TOTP code.
108    ///
109    /// Default implementation uses `load_encrypted_secret` followed by
110    /// `totp_now_from_base32`.
111    fn generate_totp_from_encrypted(
112        &self,
113        identity_reader: &mut dyn Read,
114        encrypted: &EncryptedPayload,
115    ) -> Result<String> {
116        let secret = self.load_encrypted_secret(identity_reader, encrypted)?;
117        let secret_str = std::str::from_utf8(secret.as_bytes()).map_err(|_| {
118            crate::errors::AuthError::InvalidInput("Secret is not valid UTF-8".into())
119        })?;
120        let base32 = Base32String::new(secret_str)?;
121        Self::totp_now_from_base32(&base32)
122    }
123}
