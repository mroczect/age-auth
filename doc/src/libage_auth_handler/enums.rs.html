1//! Hash algorithms supported by the OTP engine.
2//!
3//! The `Algo` enum lists every HMAC algorithm that can be used with
4//! [`crate::traits::OtpGenerator::totp`] and [`crate::traits::OtpGenerator::hotp`].
5
6use serde::{Deserialize, Serialize};
7
8/// HMAC algorithm for OTP generation.
9///
10/// # Variants
11/// - `Sha1`   – SHA‑1 (160‑bit). Legacy, not recommended for new deployments.
12/// - `Sha256` – SHA‑256 (256‑bit). **Default** and recommended choice.
13/// - `Sha512` – SHA‑512 (512‑bit). Strongest option, though no practical
14///   security benefit over SHA‑256 for OTP.
15///
16/// # Serde
17/// The enum derives `Serialize` and `Deserialize` so it can be used in
18/// configuration files or API responses.
19///
20/// # Default
21/// The default algorithm is `Sha256`. You can obtain it via `Algo::DEFAULT`
22/// or `DEFAULT_ALGO` constant.
23///
24/// # Examples
25/// ```rust
26/// use libage_auth_handler::Algo;
27///
28/// let algo = Algo::DEFAULT;
29/// assert_eq!(algo, Algo::Sha256);
30/// ```
31
32#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
33pub enum Algo {
34    /// SHA‑1 (160‑bit). Provided for compatibility. Prefer `Sha256`.
35    Sha1,
36    /// SHA‑256 (256‑bit). **Default** algorithm.
37    Sha256,
38    /// SHA‑512 (512‑bit).
39    Sha512,
40}
41
42impl Algo {
43    /// The default hash algorithm used when none is specified.
44    ///
45    /// Currently returns `Algo::Sha256`.
46    pub const DEFAULT: Algo = Algo::Sha256;
47}
