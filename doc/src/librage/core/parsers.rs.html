9pub(crate) fn parse_x25519_recipient(key: &str) -> Result<Box<dyn age::Recipient>, LibrageError> {
10    key.parse::<x25519::Recipient>()
11        .map(|r| Box::new(r) as Box<dyn age::Recipient>)
12        .map_err(|e: &'static str| LibrageError::InvalidPublicKey(e.to_string()))
13}
14
15pub(crate) fn parse_x25519_recipients(
16    keys: &[&str],
17) -> Result<Vec<Box<dyn age::Recipient>>, LibrageError> {
18    keys.iter().map(|k| parse_x25519_recipient(k)).collect()
19}

21pub(crate) fn parse_ssh_recipient(key: &str) -> Result<Box<dyn age::Recipient>, LibrageError> {
22    key.parse::<ssh::Recipient>()
23        .map(|r| Box::new(r) as Box<dyn age::Recipient>)
24        .map_err(|e: ParseRecipientKeyError| LibrageError::InvalidSshKey(format!("{:?}", e)))
25}

27pub(crate) fn parse_ssh_identity(
28    key_data: &str,
29    passphrase: Option<&str>,
30) -> Result<Box<dyn age::Identity>, LibrageError> {
31    let reader = BufReader::new(key_data.as_bytes());
32    let ssh_identity = SshIdentity::from_buffer(reader, None)
33        .map_err(|e| LibrageError::InvalidSshKey(e.to_string()))?;
34
35    match passphrase {
36        Some(pass) => match ssh_identity {
37            SshIdentity::Encrypted(enc) => {
38                let pass_secret = SecretString::new(pass.to_owned().into());
39                let unenc = enc
40                    .decrypt(pass_secret)
41                    .map_err(|e| LibrageError::InvalidSshKey(e.to_string()))?;
42                Ok(Box::new(SshIdentity::Unencrypted(unenc)) as Box<dyn age::Identity>)
43            }
44            other => Ok(Box::new(other) as Box<dyn age::Identity>),
45        },
46        None => Ok(Box::new(ssh_identity) as Box<dyn age::Identity>),
47    }
48}
